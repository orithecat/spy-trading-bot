    def _reconcile_account_positions(self, account_data):
        """Compare option contract identity and quantity, never account amounts."""
        positions = getattr(account_data, "positions", None)
        if (getattr(account_data, "positions_complete", False) is not True
                or not isinstance(positions, (tuple, list))):
            return False

        entry_totals = {}
        exit_totals = {}
        coordinator = self._order_coordinator
        orders = coordinator.orders() if coordinator is not None else []
        for order in orders:
            role = order.get("role")
            if role not in ("ENTRY", "EXIT"):
                continue
            contract = order.get("contract", {})
            key = self._contract_key(contract)
            try:
                quantity = int(order.get("filled_quantity", 0) or 0)
            except (TypeError, ValueError, ArithmeticError):
                return False
            if quantity < 0:
                return False
            if key is None:
                if quantity:
                    return False
                continue
            underlying = contract.get("underlying_symbol")
            symbol = str(contract.get("symbol", "")).upper()
            if (underlying is not None and str(underlying).upper() != "SPY"):
                if quantity:
                    return False
                continue
            if underlying is None and not symbol.startswith("SPY"):
                if quantity:
                    return False
                continue
            target = entry_totals if role == "ENTRY" else exit_totals
            target[key] = target.get(key, 0) + quantity

        expected = {}
        for key in set(entry_totals) | set(exit_totals):
            bought = entry_totals.get(key, 0)
            sold = exit_totals.get(key, 0)
            if sold > bought:
                return False
            if bought > sold:
                expected[key] = bought - sold

        actual = {}
        for position in positions:
            instrument_type = getattr(position, "instrument_type", None)
            if instrument_type not in ("EQUITY", "OPTION"):
                return False
            if instrument_type != "OPTION":
                continue
            legs = getattr(position, "legs", ())
            if not isinstance(legs, (tuple, list)):
                return False
            position_symbol = str(getattr(position, "symbol", "")).upper()
            leg_symbols = [str(getattr(leg, "symbol", "")).upper() for leg in legs]
            is_spy_position = position_symbol.startswith("SPY") or any(
                symbol.startswith("SPY") for symbol in leg_symbols
            )
            if not is_spy_position:
                continue

            contract, reason = self._emergency_position_contract(position)
            if reason or contract["underlying_symbol"] != "SPY":
                return False
            key = self._contract_key({
                "option_type": contract["option_type"],
                "strike": contract["strike"],
                "expiry": contract["expiry"],
            })
            if key is None:
                return False
            quantity = contract["quantity"]
            actual[key] = actual.get(key, 0) + quantity
        # Exact quantities reject unjournaled/manual SPY option holdings and
        # ensure holdings agree with fill-derived quantities. Equity, buying
        # power, prices, cost basis, and P/L are intentionally not compared.
        return actual == expected

    def submit_exit(self, *, position_key, contract, direction, quantity, price=None,
                    underlying_symbol=None, source="strategy1",
                    emergency_baseline_quantity=None, broker_covered_quantity=0):
        with self._lock:
            underlying_symbol = str(
                underlying_symbol or contract.get("underlying_symbol") or "SPY"
            ).upper()
            base_key = (
                underlying_symbol,
                _money(contract.get("strike_price", contract.get("strike"))).normalize(),
                str(contract.get("expiry", contract.get("option_expire_date", contract.get("expiration_date", "")))),
                str(contract.get("option_type", direction)).upper(),
            )
            covered = 0
            latest_active = None
            for order in self._orders.values():
                if order.get("role") != "EXIT" or order.get("status") not in ACTIVE_STATUSES:
                    continue
                order_contract = order.get("contract", {})
                try:
                    order_key = (
                        str(order_contract.get("underlying_symbol", "SPY")).upper(),
                        _money(order_contract.get("strike")).normalize(),
                        str(order_contract.get("expiry", "")),
                        str(order_contract.get("option_type", "")).upper(),
                    )
                except ValueError:
                    continue
                if order_key == base_key:
                    covered += max(0, int(order.get("quantity", 0)) - int(order.get("filled_quantity", 0)))

            if broker_covered_quantity < 0:
                raise OrderStateError("Covered closing quantity is invalid.")

            quantity_to_submit = max(0, int(quantity) - covered)
            if quantity_to_submit == 0:
                return dict(latest_active) if latest_active else {
                    "status": "ALREADY_COVERED", "role": "EXIT",
                    "position_key": position_key, "filled_quantity": 0,
                }

                signal_key=None, role="EXIT", direction=direction,
                contract=contract, quantity=quantity_to_submit, price=None,

    def _manage_position(self, direction, position):

                symbol=position["symbol"],
